    <!-- 2. Adicionados os scripts do Firebase --><script type="module">
        // --- PASSO A PASSO DA CONFIGURAÇÃO ---

        // PASSO 1: Configurar o Firebase
        //    1. Vá para https://console.firebase.google.com/ e crie um novo projeto.
        //    2. No seu projeto, clique no ícone da Web (</>) para adicionar uma nova aplicação web.
        //    3. Dê um nome à sua aplicação e clique em "Registar aplicação".
        //    4. O Firebase irá mostrar-lhe um objeto `firebaseConfig`. Copie esse objeto e cole-o abaixo, substituindo o objeto de exemplo.
        //    5. Vá para "Firestore Database" no menu à esquerda, clique em "Criar base de dados", inicie em "modo de produção" e escolha uma localização.
        //    6. Vá para a aba "Regras" e altere as regras para permitir leitura e escrita a utilizadores autenticados. Substitua as regras existentes por:
        //       rules_version = '2';
        //       service cloud.firestore {
        //         match /databases/{database}/documents {
        //           match /abastecimentos/{userId}/{supplyId} {
        //             allow read, write, delete: if request.auth != null && request.auth.uid == userId;
        //           }
        //         }
        //       }

        const firebaseConfig = {
            apiKey: "SUA_API_KEY",
            authDomain: "SEU_AUTH_DOMAIN",
            projectId: "SEU_PROJECT_ID",
            storageBucket: "SEU_STORAGE_BUCKET",
            messagingSenderId: "SEU_MESSAGING_SENDER_ID",
            appId: "SEU_APP_ID"
        };
        
        // PASSO 2: Configurar o Netlify
        //    1. Crie uma conta em https://www.netlify.com/
        //    2. Faça o deploy deste ficheiro HTML para um novo site no Netlify.
        //    3. No seu site Netlify, vá para "Site settings" > "Identity".
        //    4. Clique em "Enable Identity".
        //    5. Em "Registration", pode definir o registo como "Open" ou "Invite only".
        //    6. Em "External providers", pode adicionar logins com Google, GitHub, etc., se desejar.

        // --- FIM DA CONFIGURAÇÃO ---
        
        import { initializeApp } from "https://www.gstatic.com/firebasejs/9.15.0/firebase-app.js";
        import { getFirestore, collection, onSnapshot, addDoc, deleteDoc, doc, query, orderBy, serverTimestamp } from "https://www.gstatic.com/firebasejs/9.15.0/firebase-firestore.js";

        const app = initializeApp(firebaseConfig);
        const db = getFirestore(app);

        const form = document.getElementById('supply-form');
        const tableBody = document.getElementById('supply-table-body');
        const noDataMessage = document.getElementById('no-data-message');
        const contentSection = document.getElementById('content-section');
        const authSection = document.getElementById('auth-section');

        let suppliesCollectionRef;
        let unsubscribe; // Para parar de ouvir as atualizações quando o utilizador faz logout

        function setupUI(user) {
            if (user) {
                // Utilizador autenticado
                contentSection.classList.remove('hidden');
                authSection.style.display = 'none'; // Esconde a seção de auth
                
                // Configurar referência da coleção do Firestore para o utilizador atual
                suppliesCollectionRef = collection(db, 'abastecimentos', user.id, 'registros');
                
                // Escutar por mudanças em tempo real
                const q = query(suppliesCollectionRef, orderBy("data", "desc"));
                unsubscribe = onSnapshot(q, (snapshot) => {
                    const supplies = [];
                    snapshot.forEach(doc => {
                        supplies.push({ ...doc.data(), id: doc.id });
                    });
                    renderTable(supplies);
                }, (error) => {
                    console.error("Erro ao carregar dados do Firestore: ", error);
                    alert("Não foi possível carregar os dados. Verifique a configuração do Firebase e as regras de segurança.");
                });

            } else {
                // Utilizador não autenticado
                contentSection.classList.add('hidden');
                authSection.style.display = 'block';
                tableBody.innerHTML = '';
                noDataMessage.textContent = "Por favor, faça login para ver os dados.";
                noDataMessage.style.display = 'block';
                // Para a escuta de atualizações se houver uma ativa
                if (unsubscribe) unsubscribe();
            }
        }
        
        // Lógica de Autenticação Netlify
        netlifyIdentity.on('init', user => {
            setupUI(user);
        });

        netlifyIdentity.on('login', user => {
            setupUI(user);
            netlifyIdentity.close();
        });

        netlifyIdentity.on('logout', () => {
            setupUI(null);
        });


        function renderTable(data) {
            tableBody.innerHTML = '';
            if (!data || data.length === 0) {
                noDataMessage.style.display = 'block';
                tableBody.style.display = 'none';
                noDataMessage.textContent = "Nenhum dado inserido ainda.";
            } else {
                noDataMessage.style.display = 'none';
                tableBody.style.display = '';
                data.forEach(entry => {
                    const tr = document.createElement('tr');
                    tr.innerHTML = `
                        <td class="px-6 py-4 whitespace-nowrap">${entry.data || ''}</td>
                        <td class="px-6 py-4 whitespace-nowrap">${entry.prefixo || ''}</td>
                        <td class="px-6 py-4 whitespace-nowrap">${entry.placa || ''}</td>
                        <td class="px-6 py-4 whitespace-nowrap">${entry.abastecedor || ''}</td>
                        <td class="px-6 py-4 whitespace-nowrap">${entry.combustivel || ''}</td>
                        <td class="px-6 py-4 whitespace-nowrap">${entry.posto || ''}</td>
                        <td class="px-6 py-4 whitespace-nowrap text-right text-sm font-medium">
                            <button onclick="window.deleteEntry('${entry.id}')" class="text-red-600 hover:text-red-900">Excluir</button>
                        </td>
                    `;
                    tableBody.appendChild(tr);
                });
            }
        }
        
        async function addFirestoreEntry(entry) {
             try {
                await addDoc(suppliesCollectionRef, entry);
                form.reset();
            } catch (error) {
                console.error("Erro ao adicionar registro: ", error);
                alert("Não foi possível salvar o registro.");
            }
        }

        window.deleteEntry = async (id) => {
            if (!confirm(`Tem certeza que deseja excluir este registro?`)) {
                return;
            }
            try {
                const docRef = doc(db, suppliesCollectionRef.path, id);
                await deleteDoc(docRef);
            } catch(error) {
                 console.error("Erro ao excluir registro: ", error);
                 alert("Não foi possível excluir o registro.");
            }
        };

        form.addEventListener('submit', (e) => {
            e.preventDefault();
            const newEntry = {
                data: document.getElementById('data').value,
                prefixo: document.getElementById('prefixo').value,
                placa: document.getElementById('placa').value,
                abastecedor: document.getElementById('abastecedor').value,
                combustivel: document.getElementById('combustivel').value,
                posto: document.getElementById('posto').value,
            };
            addFirestoreEntry(newEntry);
        });

    </script>
</body>
